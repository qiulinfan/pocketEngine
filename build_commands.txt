# PocketEngine build/test commands

Linux configure/build:
cmake --preset unix-makefiles-debug
cmake --build --preset unix-makefiles-debug -j4

Linux release:
cmake --preset unix-makefiles-release
cmake --build --preset unix-makefiles-release -j4

Linux run runtime:
./game
# fallback (build tree location):
./build/unix-makefiles-debug/game

Linux run editor:
./pocket
# fallback (build tree location):
./build/unix-makefiles-debug/pocket

Linux refresh root compile_commands.json:
cp build/unix-makefiles-debug/compile_commands.json ./compile_commands.json
./scripts/refresh_compile_commands.sh unix-makefiles-debug

Ninja configure/build:
cmake --preset ninja-debug
cmake --build --preset ninja-debug -j4

Windows Visual Studio generate/build:
git clone https://github.com/qiulinfan/pocketEngine.git
cd .\pocketEngine
cmake --preset vs2022-x64
cmake --build --preset vs-debug
cmake --build --preset vs-release

Windows run runtime:
.\build\vs2022-x64\src\app\runtime\Debug\game.exe
.\build\vs2022-x64\src\app\runtime\Release\game.exe

Windows run editor:
.\build\vs2022-x64\src\app\editor\Debug\pocket.exe
.\build\vs2022-x64\src\app\editor\Release\pocket.exe

Windows audio smoke:
set ENGINE_LOG_AUDIO=1
set ENGINE_LOG_PERF=1
.\build\vs2022-x64\src\app\runtime\Debug\game.exe

Windows quick install:
powershell -ExecutionPolicy Bypass -File .\scripts\win_install.ps1

macOS Xcode generate/build:
cmake --preset xcode
cmake --build --preset xcode-debug
cmake --build --preset xcode-release
